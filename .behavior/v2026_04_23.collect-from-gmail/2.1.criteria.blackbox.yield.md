# blackbox criteria: collect-from-gmail

---

## episode experience

### usecase.1 = first-time setup

given('user has never authenticated with gmail')
  when('user runs gather.auth --from gmail')
    then('browser opens for google consent')
    then('user sees clear guidance on setup steps')
      sothat('user can complete oauth flow without external docs')
    then('upon consent, tokens are stored securely')
    then('user sees confirmation of successful authentication')
      sothat('user knows they can now fetch emails')

given('user has authenticated with gmail')
  when('user runs gather.auth --from gmail')
    then('extant auth is detected')
    then('user sees status of current authentication')
      sothat('user knows if re-auth is needed')

### usecase.2 = first mail sync

given('user has authenticated with gmail')
given('user has never synced emails before')
  when('user runs gather.mail --from gmail --address vlad@gmail.com')
    then('error indicates --since is required for first run')
      sothat('user explicitly controls initial sync scope')

given('user has authenticated with gmail')
given('user has never synced emails before')
  when('user runs gather.mail --from gmail --address vlad@gmail.com --since 2026-04-01')
    then('emails since the specified date are fetched')
    then('emails are stored locally')
    then('user sees count of emails synced')
    then('user sees email summaries (from, subject, date, snippet)')
      sothat('user can scan inbox without browser context-switch')

### usecase.3 = incremental sync

given('user has previously synced emails')
  when('user runs gather.mail --from gmail --address vlad@gmail.com')
    then('only new emails since last sync are fetched')
    then('user sees count of new emails')
    then('previously synced emails remain available')
      sothat('sync is efficient and does not re-fetch known emails')

given('user has previously synced emails')
given('no new emails exist')
  when('user runs gather.mail --from gmail --address vlad@gmail.com')
    then('user sees "no new emails" message')
    then('cached emails remain queryable')
      sothat('user knows sync completed without new content')

### usecase.4 = instant local lookup

given('user has synced emails previously')
  when('user runs gather.mail --from gmail --address vlad@gmail.com')
    then('response is instant (sub-second)')
    then('no network request is made for cached emails')
      sothat('user can query their mail like a local database')

### usecase.5 = token expiration and re-auth

given('user has authenticated with gmail')
given('token has expired (7-day test mode limit)')
  when('user runs gather.mail --from gmail --address vlad@gmail.com')
    then('clear error indicates token expired')
    then('error includes command to re-authenticate')
      sothat('user can quickly recover without debugging')

given('token has expired')
  when('user runs gather.auth --from gmail')
    then('new oauth flow completes')
    then('tokens are refreshed')
    then('subsequent gather.mail works')
      sothat('weekly re-auth is a single command')

## exchange experience

### usecase.6 = security-sensitive email obfuscation

given('an email contains "password" in subject or body')
  when('email is displayed')
    then('entire email content is obfuscated')
    then('only from, date, and existence are shown')
    then('obfuscation indicator is visible (e.g., lock icon)')
      sothat('terminal display does not expose security codes')

given('an email contains "mfa" or "otp" or "verification code" or "2fa" or "signin" or "reset"')
  when('email is displayed')
    then('entire email content is obfuscated')
      sothat('shoulder surfing or screen share cannot leak auth codes')

given('an email does not contain security-sensitive keywords')
  when('email is displayed')
    then('full content is shown (from, subject, date, snippet)')
      sothat('normal emails are readable')

### usecase.7 = query filters

given('user has synced emails')
  when('user runs gather.mail --from gmail --address vlad@gmail.com --query "from:boss@company.com"')
    then('only emails that match the query are returned')
    then('query syntax matches gmail web ui search')
      sothat('users can leverage familiar gmail search patterns')

given('query matches no emails')
  when('user runs gather.mail with query')
    then('empty result is returned')
    then('no error is thrown')
      sothat('empty is a valid result, not a failure')

### usecase.8 = output formats

given('user has synced emails')
  when('user runs gather.mail --from gmail --address vlad@gmail.com')
    then('output is tree format (human-readable)')
      sothat('terminal output is scannable')

given('user has synced emails')
  when('user runs gather.mail --from gmail --address vlad@gmail.com --format json')
    then('output is json (machine-readable)')
      sothat('output can pipe to jq or other tools')

### usecase.9 = address scope extension

given('user has authenticated with gmail')
  when('user runs gather.mail --from gmail --address vlad@gmail.com/labels=work,urgent')
    then('only emails with specified labels are returned')
      sothat('user can filter to specific label subsets')

### usecase.10 = error cases

given('user has not authenticated with gmail')
  when('user runs gather.mail --from gmail --address vlad@gmail.com')
    then('error indicates authentication required')
    then('error includes command to authenticate')
      sothat('user can recover without guessing')

given('invalid email address format')
  when('user runs gather.mail --from gmail --address not-an-email')
    then('error indicates invalid address format')
      sothat('user knows to fix the input')

given('user has revoked oauth access via google')
  when('user runs gather.mail')
    then('error indicates access revoked')
    then('error includes command to re-authenticate')
      sothat('user can restore access')

given('gmail api rate limit is hit')
  when('user runs gather.mail')
    then('error indicates rate limit')
    then('operation retries automatically with backoff')
    then('if retries exhausted, user sees clear error')
      sothat('transient limits are handled gracefully')

given('network timeout occurs')
  when('user runs gather.mail')
    then('operation retries automatically')
    then('if retries exhausted, user sees clear timeout error')
      sothat('transient network issues are handled gracefully')

### usecase.11 = attachments (v1 scope)

given('an email has attachments')
  when('email is displayed')
    then('attachment names are listed')
    then('attachment content is NOT downloaded')
      sothat('user sees what attachments exist without disk usage')
